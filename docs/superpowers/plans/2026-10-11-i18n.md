```ts
import { Store } from '@tauri-apps/plugin-store';

const STORE_FILE = 'preferences.json';
const LANG_KEY = 'language';
const VALID = new Set(['zh', 'en']);

let cache: Store | null = null;

async function getStore(): Promise<Store> {
  if (!cache) cache = await Store.load(STORE_FILE);
  return cache;
}

export async function getLang(): Promise<'zh' | 'en' | null> {
  try {
    const s = await getStore();
    const v = await s.get<string>(LANG_KEY);
    if (v && (VALID as Set<string>).has(v)) return v as 'zh' | 'en';
    return null;
  } catch {
    return null;
  }
}

export async function setLang(lang: 'zh' | 'en'): Promise<void> {
  const s = await getStore();
  await s.set(LANG_KEY, lang);
  await s.save();
}
```

export const LANGUAGES = ['zh', 'en'] as const;
export type Lang = (typeof LANGUAGES)[number];

export function detectBrowserLang(): Lang {
  const raw = typeof navigator !== 'undefined' ? navigator.language : 'en';
  return raw.toLowerCase().startsWith('zh') ? 'zh' : 'en';
}

export function isLang(v: unknown): v is Lang {
  return typeof v === 'string' && (LANGUAGES as readonly string[]).includes(v);
}

- [ ] **Step 1: 写 i18n/init.ts**

完整内容：

```ts
import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import { detectBrowserLang, isLang } from './languages';
import { resources } from './resources';
import { getLang, setLang } from '@/store/langStore';

const FALLBACK: 'zh' | 'en' = 'en';

async function resolveInitialLang(): Promise<'zh' | 'en'> {
  const stored = await getLang();
  if (isLang(stored)) return stored;
  return detectBrowserLang();
}

export async function initI18n(): Promise<void> {
  try {
    const lng = await resolveInitialLang();
    await i18n.use(initReactI18next).init({
      resources,
      lng,
      fallbackLng: FALLBACK,
      ns: Object.keys(resources.en),
      defaultNS: 'App',
      interpolation: { escapeValue: false },
      returnNull: false,
    });
  } catch (e) {
    console.warn('[i18n] init failed, falling back to', FALLBACK, e);
    await i18n.use(initReactI18next).init({
      resources,
      lng: FALLBACK,
      fallbackLng: FALLBACK,
      ns: Object.keys(resources.en),
      defaultNS: 'App',
      interpolation: { escapeValue: false },
      returnNull: false,
    });
  }
}

export async function changeLanguage(lang: 'zh' | 'en'): Promise<void> {
  await i18n.changeLanguage(lang);
  try {
    await setLang(lang);
  } catch (e) {
    console.warn('[i18n] failed to persist language', e);
  }
}

export { i18n };
```

  const relLabel = (type: string): string => {
